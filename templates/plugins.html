from core.plugin_api import OpenFrequencyPlugin

class Plugin(OpenFrequencyPlugin):
    """
    Main plugin class
    Must inherit from OpenFrequencyPlugin
    """

    def on_load(self):
        """Called when plugin loads"""
        self.log("Plugin loaded successfully!")

        # Read plugin config
        my_setting = self.config.get("my_setting", "default")

        # Subscribe to events
        self.subscribe("custom_event", self.handle_custom_event)

    def on_unload(self):
        """Called when plugin unloads"""
        self.log("Plugin unloading...")

    def on_atc_response(self, text, action):
        """
        ATC response hook
        Can modify ATC text
        Return None to keep original
        """
        # Example: modify specific instructions
        if "cleared" in text.lower():
            return f"鉁?{text}"
        return None

    def on_telemetry(self, data):
        """
        Telemetry hook
        ~10Hz frequency
        """
        aircraft = data.get("aircraft", {})
        altitude = aircraft.get("altitude", 0)
        speed = aircraft.get("airspeed", 0)

        # Example: log altitude changes
        if altitude > 10000:
            self.log(f"Cruising at {altitude} FT, {speed} KTS")

    def handle_custom_event(self, data):
        """Custom event handler"""
        self.log(f"Custom event received: {data}")
